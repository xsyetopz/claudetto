export type GradeFault = keyof typeof GRADE_FAULTS;

/** The grading prompt: the user's messages as the only record of what the user said, the faults in force this turn, and a JSON answer. */
export const VERIFY_GRADE = (
	people: readonly string[],
	user: string,
	reply: string,
	codes: readonly GradeFault[],
): string =>
	`You grade the final reply of a coding assistant for a fixed list of faults. Judge only the reply against the user's messages below; the user said nothing else.\n\nFaults, one code each:\n${codes.map((code) => `- ${code}: ${GRADE_FAULTS[code].test}`).join("\n")}\n\nAnswer with JSON only, no other text: {"faults":[codes found]}. Use {"faults":[]} when the reply has none of them. When unsure, answer {"faults":[]}.\n\nEarlier user messages:\n${people.map((p) => `- ${p.slice(0, 400)}`).join("\n") || "- none"}\n\nUser message that opened this turn: ${user.slice(0, 800)}\n\nAssistant reply:\n<reply>\n${reply.slice(0, 2500)}\n</reply>`;

/** The one block for the faults found: each fault's fix, in the order the faults are listed. */
export const VERIFY_BLOCK = (codes: readonly GradeFault[]): string =>
	codes.map((code) => GRADE_FAULTS[code].fix).join("\n");
